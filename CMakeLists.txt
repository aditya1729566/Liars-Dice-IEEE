cmake_minimum_required(VERSION 3.20)
project(bluff_exe LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(bluff_engine engine/src/main.cpp)
target_compile_options(bluff_engine PRIVATE -Wall -Wextra -Wpedantic)
find_package(Python3 3.12 COMPONENTS Interpreter REQUIRED)
enable_testing()
add_test(NAME engine_rules COMMAND ${Python3_EXECUTABLE} -m unittest discover -s engine/tests -v)
set_tests_properties(engine_rules PROPERTIES WORKING_DIRECTORY ${CMAKE_SOURCE_DIR})
